你这里确实到了一个容易混的地方。**从 send_packet() 这一层开始，最大的误区是：你现在手里已经有“PNG 二进制数据”，不需要再自己创建一个 Packet\* 去接收 send_packet() 的返回值。**

你原来这句：

```
Packet* packet = (Packet*)send_packet(cfd, CMD_SCREEN, pdata, len);
```

是**错的**。

因为我们之前设计的 `send_packet()` 返回的是：

```
0   -> 发送成功
-1  -> 发送失败
```

它返回的不是 `Packet*`。

所以这里应该是：

```
int ret = send_packet(cfd, CMD_SCREEN, pdata, len);

if (ret != 0)
{
    printf("发送屏幕数据失败\n");
}
```

另外，你这里还有一个非常重要的问题：

```
int len = GlobalSize(hMEN);
```

**不能直接把 GlobalSize() 当成 PNG 实际数据长度。**

因为 `GlobalSize()` 得到的是这块 HGLOBAL 的分配大小，不一定等于 PNG 实际写入流的长度。

我们应该：

```
CImage
 ↓
PNG保存到IStream
 ↓
得到“流实际有多少字节”
 ↓
把流指针移动到开头
 ↓
GlobalLock()
 ↓
得到PNG二进制数据指针
 ↓
send_packet()
 ↓
send_exact()
```

你现在这一段，我建议这样写。

## `HandleScreen()` 修改后的完整版本

```
int HandleScreen(SOCKET cfd, Packet* packet)
{
    // packet这个参数现在暂时没有使用。
    // 因为我们发送PNG时，会在send_packet()里面自己创建Packet。
    // 所以这里先保留这个参数，方便以后扩展。


    // 获取本地屏幕

    CImage image;


    // 拿到屏幕DC
    // 可以理解成：向Windows借一个“操作屏幕”的工具

    HDC hdc = GetDC(NULL);

    if (hdc == NULL)
    {
        printf("获取屏幕DC失败\n");
        return -1;
    }


    // 获取屏幕像素深度

    int bitdepth =
        GetDeviceCaps(
            hdc,
            BITSPIXEL
        );


    // 设置DPI感知
    //
    // 告诉Windows：
    // 这个程序自己处理真实像素，
    // 不要帮我进行缩放。

    SetProcessDpiAwarenessContext(
        DPI_AWARENESS_CONTEXT_PER_MONITOR_AWARE_V2
    );


    // 获取屏幕宽度

    int sWidth =
        GetSystemMetrics(SM_CXSCREEN);


    // 获取屏幕高度

    int sHeight =
        GetSystemMetrics(SM_CYSCREEN);


    printf(
        "屏幕宽度：%d，高度：%d，像素深度：%d\n",
        sWidth,
        sHeight,
        bitdepth
    );


    // 创建一个和屏幕一样大的CImage

    image.Create(
        sWidth,
        sHeight,
        bitdepth
    );


    // 把屏幕内容复制到image中

    BitBlt(
        image.GetDC(),     // 目标：image
        0,
        0,
        sWidth,
        sHeight,
        hdc,               // 来源：屏幕
        0,
        0,
        SRCCOPY
    );


    // 释放屏幕DC

    ReleaseDC(
        NULL,
        hdc
    );


    // =========================================================
    // 下面开始：
    // CImage
    //    ↓
    // PNG
    //    ↓
    // 内存
    //    ↓
    // 网络
    // =========================================================


    // 创建一个HGLOBAL
    //
    // 一开始大小是0。
    //
    // 后面IStream保存PNG时，
    // Windows会根据需要自动扩展这块内存。

    HGLOBAL hMEN =
        GlobalAlloc(
            GMEM_MOVEABLE,
            0
        );

    if (hMEN == NULL)
    {
        printf("GlobalAlloc失败\n");
        return -1;
    }


    // 创建一个绑定到hMEN的内存流
    //
    // TRUE表示：
    // pStream释放的时候，
    // hMEN也会自动释放。
    //
    // 所以后面不要再自己GlobalFree(hMEN)。

    IStream* pStream = NULL;

    HRESULT hr =
        CreateStreamOnHGlobal(
            hMEN,
            TRUE,
            &pStream
        );

    if (FAILED(hr))
    {
        printf("CreateStreamOnHGlobal失败\n");

        // 因为CreateStreamOnHGlobal失败，
        // 这时候流没有接管hMEN，
        // 所以需要自己释放hMEN。

        GlobalFree(hMEN);

        return -1;
    }


    // =========================================================
    // 第1步：把CImage保存成PNG
    //
    // 原来屏幕是：
    //
    // 像素数据
    //
    // 现在变成：
    //
    // PNG二进制数据
    //
    // 保存的位置不是文件，
    // 而是pStream对应的内存。
    // =========================================================

    hr =
        image.Save(
            pStream,
            Gdiplus::ImageFormatPNG
        );

    if (FAILED(hr))
    {
        printf("CImage保存PNG失败\n");

        pStream->Release();

        return -1;
    }


    // =========================================================
    // 第2步：获取PNG实际长度
    //
    // 这里不要使用：
    //
    // GlobalSize(hMEN)
    //
    // 因为GlobalSize得到的是HGLOBAL的分配大小，
    // 不一定等于PNG实际数据长度。
    //
    // 我们应该问IStream：
    // “你现在里面实际有多少数据？”
    // =========================================================

    LARGE_INTEGER moveEnd;

    moveEnd.QuadPart = 0;

    ULARGE_INTEGER streamSize;

    hr =
        pStream->Seek(
            moveEnd,
            STREAM_SEEK_END,
            &streamSize
        );

    if (FAILED(hr))
    {
        printf("获取PNG长度失败\n");

        pStream->Release();

        return -1;
    }


    // streamSize.QuadPart就是：
    //
    // PNG实际有多少字节

    ULONGLONG imageSize =
        streamSize.QuadPart;


    // =========================================================
    // 第3步：检查长度
    //
    // send_packet使用uint32_t保存长度，
    // 所以这里不能超过uint32_t能表示的范围。
    //
    // 同时还需要考虑你的Packet.body容量。
    // =========================================================

    if (imageSize == 0)
    {
        printf("PNG数据为空\n");

        pStream->Release();

        return -1;
    }

    if (imageSize >= BUF_SIZE)
    {
        printf(
            "屏幕图片太大：%llu 字节\n",
            imageSize
        );

        pStream->Release();

        return -1;
    }


    // =========================================================
    // 第4步：把流指针重新移动到开头
    //
    // 前面我们刚刚Seek到末尾了。
    //
    // 现在准备读取PNG数据，
    // 所以必须回到第0字节。
    // =========================================================

    LARGE_INTEGER moveBegin;

    moveBegin.QuadPart = 0;

    hr =
        pStream->Seek(
            moveBegin,
            STREAM_SEEK_SET,
            NULL
        );

    if (FAILED(hr))
    {
        printf("流回到开头失败\n");

        pStream->Release();

        return -1;
    }


    // =========================================================
    // 第5步：把HGLOBAL转换成普通指针
    //
    // hMEN：
    //     是一个“内存句柄”
    //
    // GlobalLock：
    //     把句柄转换成真正可以访问的内存地址。
    //
    // 最终：
    //
    // pdata
    //    ↓
    // PNG二进制数据
    // =========================================================

    char* pdata =
        (char*)GlobalLock(hMEN);

    if (pdata == NULL)
    {
        printf("GlobalLock失败\n");

        pStream->Release();

        return -1;
    }


    // =========================================================
    // 第6步：把PNG数据交给send_packet()
    //
    // 注意！
    //
    // 这里千万不要写：
    //
    // Packet* packet = (Packet*)send_packet(...);
    //
    // 因为send_packet返回的是：
    //
    // 0  -> 成功
    // -1 -> 失败
    //
    // 它不是Packet*。
    //
    // 我们这里只需要接收一个int返回值。
    // =========================================================

    int ret =
        send_packet(
            cfd,                       // 客户端socket
            CMD_SCREEN,                // 命令号：屏幕截图
            pdata,                     // PNG二进制数据
            (uint32_t)imageSize        // PNG真实长度
        );


    // =========================================================
    // 第7步：解除内存锁定
    //
    // GlobalLock()之后，
    // 对应需要GlobalUnlock()。
    // =========================================================

    GlobalUnlock(hMEN);


    // =========================================================
    // 第8步：释放IStream
    //
    // 因为创建IStream时：
    //
    // fDeleteOnRelease = TRUE
    //
    // 所以这里Release以后，
    // IStream会自动释放它管理的HGLOBAL。
    //
    // 不要再：
    //
    // GlobalFree(hMEN);
    // =========================================================

    pStream->Release();


    // =========================================================
    // 第9步：检查发送结果
    // =========================================================

    if (ret != 0)
    {
        printf(
            "发送屏幕截图失败\n"
        );

        return -1;
    }


    printf(
        "屏幕截图发送成功，大小：%llu 字节\n",
        imageSize
    );

    return 0;
}
```

 ↓

保证全部发送

接收完整数据
